/* =====================================================================
   RECEBE OS LEADS DA LANDING PAGE
   ===================================================================== */
function doPost(e) {
  var lock = LockService.getScriptLock();
  var travou = lock.tryLock(10000);
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return saida_({ status: 'erro', msg: 'Sem dados. Esta função é chamada pelo site.' });
    }
    var d = JSON.parse(e.postData.contents);
    var sh = abaLeads_();

    var whats = String(d.whatsapp || '').replace(/\D/g, '');
    if ((whats.length === 12 || whats.length === 13) && whats.indexOf('55') === 0) whats = whats.slice(2);
    var nome = limpar_(d.nome);

    sh.appendRow([
      new Date(),
      nome,
      formatarFone_(whats),
      '',
      'Novo',
      '',
      limpar_(d.origem) || 'Landing page',
      limpar_(d.campanha) || '(sem campanha)',
      limpar_(d.fonte) || '(direto)',
      limpar_(d.termo),
      limpar_(d.gclid),
      limpar_(d.pagina)
    ]);

    var r = sh.getLastRow();
    if (whats) {
      var texto = MENSAGEM_RETORNO.replace('{nome}', primeiroNome_(nome));
      var link = SpreadsheetApp.newRichTextValue()
        .setText('💬  Chamar no WhatsApp')
        .setLinkUrl('https://wa.me/55' + whats + '?text=' + encodeURIComponent(texto))
        .build();
      sh.getRange(r, C.CONVERSA).setRichTextValue(link);
    }
    sh.getRange(r, C.DATA).setNumberFormat('dd/MM/yyyy HH:mm');
    return saida_({ status: 'ok' });
  } catch (err) {
    return saida_({ status: 'erro', msg: String(err) });
  } finally {
    if (travou) lock.releaseLock();
  }
}

function doGet() {

}

/* =====================================================================
   MONTA A PLANILHA INTEIRA (rodar uma vez)
   ===================================================================== */
function configurarPlanilha() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.setSpreadsheetTimeZone('America/Bahia');
  try { ss.setSpreadsheetLocale('pt_BR'); } catch (e) {}
  ss.rename('Leads ' + CLINICA + ' · LP Implantes');

  var leads = montarLeads_(ss);
  var dados = montarDados_(ss);
  var painel = montarPainel_(ss, dados);

  // ordem das abas: Painel, Leads, (Dados escondida)
  ss.setActiveSheet(painel); ss.moveActiveSheet(1);
  ss.setActiveSheet(leads); ss.moveActiveSheet(2);
  dados.hideSheet();
  ss.getSheets().forEach(function (s) {
    if (['Painel', 'Leads', 'Dados'].indexOf(s.getName()) === -1 && s.getLastRow() === 0) ss.deleteSheet(s);
  });
  ss.setActiveSheet(painel);
  SpreadsheetApp.flush();
}

/* ---------- Aba LEADS ---------- */
function montarLeads_(ss) {
  var sh = ss.getSheetByName('Leads');
  if (!sh) {
    var vazia = ss.getSheets().filter(function (s) { return s.getLastRow() === 0 && s.getName() !== 'Painel' && s.getName() !== 'Dados'; })[0];
    sh = vazia ? vazia.setName('Leads') : ss.insertSheet('Leads');
  }
  var n = CABECALHO.length;
  if (sh.getMaxRows() < 1000) sh.insertRowsAfter(sh.getMaxRows(), 1000 - sh.getMaxRows());
  var max = sh.getMaxRows();
  if (sh.getMaxColumns() < n) sh.insertColumnsAfter(sh.getMaxColumns(), n - sh.getMaxColumns());
  if (sh.getMaxColumns() > n) sh.deleteColumns(n + 1, sh.getMaxColumns() - n);

  sh.setTabColor(COR.royal);
  sh.setHiddenGridlines(true);
  sh.getRange(1, 1, 1, n).setValues([CABECALHO]);

  var tudo = sh.getRange(1, 1, max, n);
  tudo.setFontFamily(FONTE).setFontSize(10).setFontColor(COR.texto).setVerticalAlignment('middle');

  var cab = sh.getRange(1, 1, 1, n);
  cab.setFontWeight('bold').setFontSize(10).setFontColor(COR.branco).setBackground(COR.marinho)
     .setHorizontalAlignment('center').setWrap(true);
  sh.setRowHeight(1, 44);
  sh.setRowHeights(2, max - 1, 32);
  sh.setFrozenRows(1);
  sh.setFrozenColumns(2);
  LARGURAS.forEach(function (w, i) { sh.setColumnWidth(i + 1, w); });

  // dicas que aparecem ao passar o mouse no cabeçalho
  sh.getRange(1, C.WHATS).setNote('Fundo laranja = este número já se cadastrou antes.');
  sh.getRange(1, C.CONVERSA).setNote('Clique no link para abrir a conversa no WhatsApp com uma mensagem pronta.');
  sh.getRange(1, C.STATUS).setNote('Clique na seta para atualizar o status do atendimento. O Painel se atualiza sozinho.');
  sh.getRange(1, C.OBS).setNote('Espaço livre para anotações da equipe (ex.: preferiu horário à tarde).');

  // formatos das colunas
  sh.getRange(2, C.DATA, max - 1, 1).setNumberFormat('dd/MM/yyyy HH:mm').setHorizontalAlignment('center');
  sh.getRange(2, C.NOME, max - 1, 1).setFontWeight('bold');
  sh.getRange(2, C.WHATS, max - 1, 1).setHorizontalAlignment('center');
  sh.getRange(2, C.CONVERSA, max - 1, 2).setHorizontalAlignment('center');
  sh.getRange(2, C.OBS, max - 1, 1).setFontColor(COR.texto).setFontStyle('italic');
  sh.getRange(2, C.ORIGEM, max - 1, 6).setFontColor(COR.cinza).setFontSize(9);

  // listras azul-claro
  sh.getBandings().forEach(function (b) { b.remove(); });
  tudo.applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, true, false)
      .setHeaderRowColor(COR.marinho).setFirstRowColor(COR.branco).setSecondRowColor(COR.gelo2);

  // status com lista suspensa
  var regraLista = SpreadsheetApp.newDataValidation()
      .requireValueInList(STATUS.map(function (s) { return s[0]; }), true)
      .setAllowInvalid(false).build();
  sh.getRange(2, C.STATUS, max - 1, 1).setDataValidation(regraLista).setFontWeight('bold');

  // regras de cor: número repetido, status e leads de hoje
  var rStatus = sh.getRange(2, C.STATUS, max - 1, 1);
  var regras = [];
  regras.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($C2<>"",COUNTIF($C$2:$C,$C2)>1)')
    .setBackground('#FFE9C7').setFontColor('#8A5A00').setRanges([sh.getRange(2, C.NOME, max - 1, 2)]).build());
  STATUS.forEach(function (s) {
    regras.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(s[0])
      .setBackground(s[1]).setFontColor(s[2]).setRanges([rStatus]).build());
  });
  regras.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($A2<>"",$A2>=TODAY())')
    .setBackground('#E3ECFF').setRanges([sh.getRange(2, C.DATA, max - 1, 3)]).build());
  sh.setConditionalFormatRules(regras);

  if (!sh.getFilter()) sh.getRange(1, 1, max, n).createFilter();
  sh.getRange(1, 1, max, n).setBorder(null, null, null, null, false, true, COR.linha, SpreadsheetApp.BorderStyle.SOLID);
  return sh;
}

/* ---------- Aba DADOS (escondida, alimenta os gráficos) ---------- */
function montarDados_(ss) {
  var sh = ss.getSheetByName('Dados') || ss.insertSheet('Dados');
  sh.clear();
  sh.getCharts().forEach(function (c) { sh.removeChart(c); });

  // leads por dia (14 dias)
  sh.getRange('A1:C1').setValues([['Dia', 'Leads', 'Data']]);
  var f = [];
  for (var i = 0; i < 14; i++) {
    var lin = i + 2;
    f.push([
      '=TEXT(C' + lin + ',"dd/mm")',
      '=COUNTIFS(Leads!$A$2:$A,">="&C' + lin + ',Leads!$A$2:$A,"<"&(C' + lin + '+1))',
      '=TODAY()-' + (13 - i)
    ]);
  }
  sh.getRange(2, 1, 14, 3).setFormulas(f);

  // últimos 10 leads (data, nome, whatsapp, status)
  sh.getRange('E1').setValue('Últimos leads');
  sh.getRange('E2').setFormula('=IFERROR(QUERY({Leads!A2:C,Leads!E2:E},"select Col1,Col2,Col3,Col4 where Col2 is not null order by Col1 desc limit 10",0),"")');

  // por botão de origem (coluna G)
  sh.getRange('J1').setValue('Por botão');
  sh.getRange('J2').setFormula('=IFERROR(QUERY({Leads!G2:G},"select Col1, count(Col1) where Col1 is not null group by Col1 order by count(Col1) desc label Col1 \'Botão\', count(Col1) \'Leads\'",0),{"Botão","Leads";"Aguardando leads",0})');

  // por campanha (coluna H)
  sh.getRange('M1').setValue('Por campanha');
  sh.getRange('M2').setFormula('=IFERROR(QUERY({Leads!H2:H},"select Col1, count(Col1) where Col1 is not null group by Col1 order by count(Col1) desc label Col1 \'Campanha\', count(Col1) \'Leads\'",0),{"Campanha","Leads";"Aguardando leads",0})');
  return sh;
}

}
function titulo_(sh, a1, texto) {
  sh.getRange(a1).merge().setValue(texto).setFontSize(14).setFontWeight('bold')
    .setFontColor(COR.marinho).setHorizontalAlignment('left')
    .setBorder(null, null, true, null, null, null, COR.royal, SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
}
function limpar_(v) {
  var s = String(v == null ? '' : v).trim().slice(0, 300);
  if (/^[=+\-@]/.test(s)) s = "'" + s; // evita fórmulas injetadas
  return s;
}
function primeiroNome_(nome) {
  var p = String(nome || '').replace(/^'/, '').trim().split(/\s+/)[0] || '';
  return p ? p.charAt(0).toUpperCase() + p.slice(1).toLowerCase() : 'tudo bem';
}
function formatarFone_(d) {
  if (d.length === 11) return '(' + d.slice(0, 2) + ') ' + d.slice(2, 7) + '-' + d.slice(7);
  if (d.length === 10) return '(' + d.slice(0, 2) + ') ' + d.slice(2, 6) + '-' + d.slice(6);
  return d;
}

}
