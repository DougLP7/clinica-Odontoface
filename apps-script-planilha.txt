/* ---------------------------------------------------------------------
   RECEBE OS LEADS DA LANDING PAGE
   --------------------------------------------------------------------- */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return saida_({ status: 'erro', msg: 'Sem dados. Esta função é chamada pelo site.' });
    }
    var d = JSON.parse(e.postData.contents);
    var sh = abaLeads_();
    var whats = String(d.whatsapp || '').replace(/\D/g, '');
    if (whats.length === 12 || whats.length === 13) { if (whats.indexOf('55') === 0) whats = whats.slice(2); }

    sh.appendRow([
      new Date(),
      limpar_(d.nome),
      formatarFone_(whats),
      '',
      'Novo',
      limpar_(d.origem) || 'Landing page',
      limpar_(d.campanha) || '(sem campanha)',
      limpar_(d.fonte) || '(direto)',
      limpar_(d.termo),
      limpar_(d.gclid),
      limpar_(d.pagina)
    ]);

    var r = sh.getLastRow();
    if (whats) {
      var link = SpreadsheetApp.newRichTextValue()
        .setText('💬 Chamar no WhatsApp')
        .setLinkUrl('https://wa.me/55' + whats)
        .build();
      sh.getRange(r, 4).setRichTextValue(link);
    }
    sh.getRange(r, 1).setNumberFormat('dd/MM/yyyy HH:mm');
    return saida_({ status: 'ok' });
  } catch (err) {
    return saida_({ status: 'erro', msg: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  return saida_({ status: 'ok', msg: 'Planilha de leads da Odontoface ativa.' });
}

/* ---------------------------------------------------------------------
   MONTA A PLANILHA INTEIRA (rodar uma vez)
   --------------------------------------------------------------------- */
function configurarPlanilha() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.setSpreadsheetTimeZone('America/Sao_Paulo');
  try { ss.setSpreadsheetLocale('pt_BR'); } catch (e) {}
  ss.rename('Leads Odontoface · LP Implantes');

  var leads = montarLeads_(ss);
  var dados = montarDados_(ss);
  var painel = montarPainel_(ss, dados);

  // ordem das abas: Painel, Leads, (Dados escondida)
  ss.setActiveSheet(painel); ss.moveActiveSheet(1);
  ss.setActiveSheet(leads); ss.moveActiveSheet(2);
  dados.hideSheet();
  ss.getSheets().forEach(function (s) {
    if (['Painel', 'Leads', 'Dados'].indexOf(s.getName()) === -1 && s.getLastRow() === 0) ss.deleteSheet(s);
  });
  ss.setActiveSheet(painel);
  SpreadsheetApp.flush();
}

/* ---------- Aba LEADS ---------- */
function montarLeads_(ss) {
  var sh = ss.getSheetByName('Leads');
  if (!sh) {
    var vazia = ss.getSheets().filter(function (s) { return s.getLastRow() === 0 && s.getName() !== 'Painel' && s.getName() !== 'Dados'; })[0];
    sh = vazia ? vazia.setName('Leads') : ss.insertSheet('Leads');
  }
  var max = Math.max(sh.getMaxRows(), 1000);
  if (sh.getMaxRows() < 1000) sh.insertRowsAfter(sh.getMaxRows(), 1000 - sh.getMaxRows());
  if (sh.getMaxColumns() < CABECALHO.length) sh.insertColumnsAfter(sh.getMaxColumns(), CABECALHO.length - sh.getMaxColumns());
  if (sh.getMaxColumns() > CABECALHO.length) sh.deleteColumns(CABECALHO.length + 1, sh.getMaxColumns() - CABECALHO.length);

  sh.setTabColor(CORES.vinho);
  sh.getRange(1, 1, 1, CABECALHO.length).setValues([CABECALHO]);

  var tudo = sh.getRange(1, 1, max, CABECALHO.length);
  tudo.setFontFamily(FONTE_TEXTO).setFontSize(10).setFontColor(CORES.texto).setVerticalAlignment('middle');

  var cab = sh.getRange(1, 1, 1, CABECALHO.length);
  cab.setFontWeight('bold').setFontSize(10).setFontColor(CORES.branco).setBackground(CORES.vinho)
     .setHorizontalAlignment('center').setWrap(true);
  sh.setRowHeight(1, 42);
  sh.setRowHeights(2, max - 1, 30);
  sh.setFrozenRows(1);
  sh.setFrozenColumns(2);

  var larguras = [140, 210, 150, 190, 140, 210, 170, 140, 140, 150, 240];
  larguras.forEach(function (w, i) { sh.setColumnWidth(i + 1, w); });

  sh.getRange(2, 1, max - 1, 1).setNumberFormat('dd/MM/yyyy HH:mm').setHorizontalAlignment('center');
  sh.getRange(2, 2, max - 1, 1).setFontWeight('bold');
  sh.getRange(2, 3, max - 1, 1).setHorizontalAlignment('center');
  sh.getRange(2, 4, max - 1, 2).setHorizontalAlignment('center');
  sh.getRange(2, 7, max - 1, 5).setFontColor(CORES.cinza).setFontSize(9);

  // listras nas cores da marca
  sh.getBandings().forEach(function (b) { b.remove(); });
  tudo.applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, true, false)
      .setHeaderRowColor(CORES.vinho).setFirstRowColor(CORES.branco).setSecondRowColor('#F5F8FD');

  // status com lista suspensa
  var regraLista = SpreadsheetApp.newDataValidation()
      .requireValueInList(STATUS.map(function (s) { return s[0]; }), true)
      .setAllowInvalid(false).build();
  sh.getRange(2, 5, max - 1, 1).setDataValidation(regraLista).setFontWeight('bold');

  // cores do status + destaque para leads de hoje
  var rStatus = sh.getRange(2, 5, max - 1, 1);
  var regras = STATUS.map(function (s) {
    return SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(s[0])
      .setBackground(s[1]).setFontColor(s[2]).setRanges([rStatus]).build();
  });
  regras.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($A2<>"",$A2>=TODAY())')
    .setBackground('#FFF8E6').setRanges([sh.getRange(2, 1, max - 1, 3)]).build());
  sh.setConditionalFormatRules(regras);

  if (!sh.getFilter()) sh.getRange(1, 1, max, CABECALHO.length).createFilter();
  sh.getRange(1, 1, max, CABECALHO.length).setBorder(null, null, null, null, false, true, CORES.linha, SpreadsheetApp.BorderStyle.SOLID);
  return sh;
}

/* ---------- Aba DADOS (escondida, alimenta os gráficos) ---------- */
function montarDados_(ss) {
  var sh = ss.getSheetByName('Dados') || ss.insertSheet('Dados');
  sh.clear();
  sh.getCharts().forEach(function (c) { sh.removeChart(c); });

  // leads por dia (14 dias)
  sh.getRange('A1:C1').setValues([['Dia', 'Leads', 'Data']]);
  var f = [];
  for (var i = 0; i < 14; i++) {
    var lin = i + 2;
    f.push([
      '=TEXT(C' + lin + ',"dd/mm")',
      '=COUNTIFS(Leads!$A$2:$A,">="&C' + lin + ',Leads!$A$2:$A,"<"&(C' + lin + '+1))',
      '=TODAY()-' + (13 - i)
    ]);
  }
  sh.getRange(2, 1, 14, 3).setFormulas(f);

  // últimos 10 leads
  sh.getRange('E1').setValue('Últimos leads');
  sh.getRange('E2').setFormula('=IFERROR(QUERY({Leads!A2:C,Leads!E2:E},"select Col1,Col2,Col3,Col4 where Col2 is not null order by Col1 desc limit 10",0),"")');

  // por botão de origem
  sh.getRange('J1').setValue('Por botão');
  sh.getRange('J2').setFormula('=IFERROR(QUERY({Leads!F2:F},"select Col1, count(Col1) where Col1 is not null group by Col1 order by count(Col1) desc label Col1 \'Botão\', count(Col1) \'Leads\'",0),{"Botão","Leads";"Aguardando leads",0})');

  // por campanha
  sh.getRange('M1').setValue('Por campanha');
  sh.getRange('M2').setFormula('=IFERROR(QUERY({Leads!G2:G},"select Col1, count(Col1) where Col1 is not null group by Col1 order by count(Col1) desc label Col1 \'Campanha\', count(Col1) \'Leads\'",0),{"Campanha","Leads";"Aguardando leads",0})');
  return sh;
}

/* ---------- auxiliares ---------- */
function abaLeads_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('Leads');
  if (!sh) { configurarPlanilha(); sh = ss.getSheetByName('Leads'); }
  return sh;
}
function titulo_(sh, a1, texto) {
  sh.getRange(a1).merge().setValue(texto).setFontFamily(FONTE_TITULO).setFontSize(14).setFontWeight('bold')
    .setFontColor(CORES.vinho).setHorizontalAlignment('left')
    .setBorder(null, null, true, null, null, null, CORES.vinho3, SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
}
function limpar_(v) {
  var s = String(v == null ? '' : v).trim().slice(0, 300);
  if (/^[=+\-@]/.test(s)) s = "'" + s; // evita fórmulas injetadas
  return s;
}
function formatarFone_(d) {
  if (d.length === 11) return '(' + d.slice(0, 2) + ') ' + d.slice(2, 7) + '-' + d.slice(7);
  if (d.length === 10) return '(' + d.slice(0, 2) + ') ' + d.slice(2, 6) + '-' + d.slice(6);
  return d;
}
function saida_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
